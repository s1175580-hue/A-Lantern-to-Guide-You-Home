    /* =========================================================================
       【素材替换区】当你从 Procreate 导出真实 PNG 后，只需把路径填入下面 4 个常量：
       ========================================================================= */
    const BG_INITIAL_SRC     = 'bg_initial.png'; // 例如: 'images/bg_initial.png'      (图1去掉白色鸟笼后的背景)
    const CAGE_PNG_SRC       = 'cage.png'; // 例如: 'images/cage.png'            (图1单独导出的透明背景白色鸟笼)
    const FOUNTAIN_FRONT_SRC = 'fountain.png'; // 例如: 'images/fountain_front.png'  (可选：遮在鸟笼下方的蓝色水流透明图)
    const BG_HEALED_SRC      = 'bg_healed.png'; // 例如: 'images/bg_healed.png'       (图2：金色月亮 + 发光白乌鸦 + 石板路)

    /* =========================================================================
       【程序化高清 Procreate 风格美术生成器】
       在未填入外部 PNG 路径时，自动绘制与你上传的两张 Procreate 画作高度一致的 4 层美术素材
       ========================================================================= */
    const ART_W = 1600;
    const ART_H = 1200;

    // 辅助函数：绘制共用的暗炭蓝纹理夜空、藤蔓线稿、等高线蓝山与白色石喷泉底座
    function drawSharedLandscape(ctx, isHealed) {
      // 1. 深炭蓝纸纹底色
      ctx.fillStyle = '#141d29';
      ctx.fillRect(0, 0, ART_W, ART_H);

      // 细腻纸张噪点质感
      ctx.save();
      ctx.fillStyle = 'rgba(255, 255, 255, 0.025)';
      for (let i = 0; i < 1800; i++) {
        ctx.fillRect((i * 197) % ART_W, (i * 313) % ART_H, 2, 2);
      }
      ctx.restore();

      // 2. 背景暗绿灰藤蔓与叶片暗纹
      ctx.save();
      ctx.strokeStyle = 'rgba(130, 155, 150, 0.22)';
      ctx.fillStyle = 'rgba(130, 155, 150, 0.14)';
      ctx.lineWidth = 2.5;
      const vines = [
        { sx: 80, sy: 0, cx1: 220, cy1: 260, cx2: 40, cy2: 480, ex: 190, ey: 720 },
        { sx: 460, sy: 0, cx1: 350, cy1: 180, cx2: 520, cy2: 340, ex: 410, ey: 520 },
        { sx: 1180, sy: 0, cx1: 1320, cy1: 200, cx2: 1120, cy2: 380, ex: 1260, ey: 560 },
        { sx: 1510, sy: 0, cx1: 1370, cy1: 250, cx2: 1540, cy2: 470, ex: 1410, ey: 710 }
      ];
      vines.forEach(v => {
        ctx.beginPath();
        ctx.moveTo(v.sx, v.sy);
        ctx.bezierCurveTo(v.cx1, v.cy1, v.cx2, v.cy2, v.ex, v.ey);
        ctx.stroke();

        // 沿藤蔓点缀叶片
        for (let t = 0.15; t <= 0.85; t += 0.16) {
          const lx = (1 - t) * (1 - t) * v.sx + 2 * (1 - t) * t * v.cx1 + t * t * v.ex;
          const ly = (1 - t) * (1 - t) * v.sy + 2 * (1 - t) * t * v.cy1 + t * t * v.ey;
          const dir = Math.round(t * 10) % 2 === 0 ? 1 : -1;
          ctx.save();
          ctx.translate(lx, ly);
          ctx.rotate(dir * 0.55);
          ctx.beginPath();
          ctx.ellipse(dir * 24, 0, 26, 11, 0, 0, Math.PI * 2);
          ctx.fill();
          ctx.stroke();
          ctx.restore();
        }
      });
      ctx.restore();

      // 3. 图2专属：升起的温暖金色大弯月
      if (isHealed) {
        ctx.save();
        ctx.shadowBlur = 45;
        ctx.shadowColor = 'rgba(234, 214, 128, 0.55)';
        ctx.fillStyle = '#e4d587';
        ctx.beginPath();
        ctx.arc(800, 350, 355, 0.08 * Math.PI, 0.92 * Math.PI, false);
        ctx.bezierCurveTo(510, 560, 510, 210, 550, 115);
        ctx.bezierCurveTo(585, 390, 1015, 390, 1050, 115);
        ctx.bezierCurveTo(1090, 210, 1090, 560, 1140, 440);
        ctx.closePath();
        // 使用双圆相减绘制标准饱满月牙
        ctx.restore();

        ctx.save();
        const moonCanvas = document.createElement('canvas');
        moonCanvas.width = ART_W;
        moonCanvas.height = ART_H;
        const mCtx = moonCanvas.getContext('2d');
        mCtx.fillStyle = '#e5d68a';
        mCtx.beginPath();
        mCtx.arc(800, 360, 365, 0, Math.PI * 2);
        mCtx.fill();
        mCtx.globalCompositeOperation = 'destination-out';
        mCtx.beginPath();
        mCtx.arc(800, 250, 330, 0, Math.PI * 2);
        mCtx.fill();

        ctx.shadowBlur = 35;
        ctx.shadowColor = 'rgba(235, 220, 140, 0.5)';
        ctx.drawImage(moonCanvas, 0, 0);
        ctx.restore();
      }

      // 4. 几何三角山峰（带等高线纹理与暗面切面，还原 Procreate 山峦）
      function drawMountain(peakX, peakY, leftX, rightX, baseY, fillMain, fillShadow, hasContour) {
        ctx.save();
        // 主山体
        ctx.fillStyle = fillMain;
        ctx.beginPath();
        ctx.moveTo(peakX, peakY);
        ctx.lineTo(leftX, baseY);
        ctx.lineTo(rightX, baseY);
        ctx.closePath();
        ctx.fill();

        // 等高线波纹装饰
        if (hasContour) {
          ctx.save();
          ctx.clip();
          ctx.strokeStyle = 'rgba(180, 205, 235, 0.25)';
          ctx.lineWidth = 1.6;
          for (let r = 40; r < 520; r += 26) {
            ctx.beginPath();
            ctx.ellipse(peakX - 30, peakY + r * 0.85, r * 0.75, r * 0.48, 0.2, 0, Math.PI * 2);
            ctx.stroke();
          }
          ctx.restore();
        }

        // 右侧暗面切面
        ctx.fillStyle = fillShadow;
        ctx.beginPath();
        ctx.moveTo(peakX, peakY);
        ctx.lineTo(peakX + (rightX - peakX) * 0.18, baseY);
        ctx.lineTo(rightX, baseY);
        ctx.closePath();
        ctx.fill();

        // 细白描边
        ctx.strokeStyle = 'rgba(220, 232, 245, 0.45)';
        ctx.lineWidth = 1.8;
        ctx.beginPath();
        ctx.moveTo(leftX, baseY);
        ctx.lineTo(peakX, peakY);
        ctx.lineTo(rightX, baseY);
        ctx.stroke();
        ctx.restore();
      }

      // 远景山峰
      drawMountain(275, 445, 30, 520, 960, '#22385c', '#0e1726', true);
      drawMountain(1150, 520, 920, 1380, 960, '#1c2f4d', '#0b121e', false);
      drawMountain(1435, 415, 1180, 1660, 960, '#243c63', '#101929', true);

      // 近景山峰（左峰带等高线与亮灰切面，右峰带深黑切面）
      drawMountain(520, 355, 220, 820, 980, '#274069', '#090e17', true);
      drawMountain(115, 625, -40, 290, 980, '#2b4673', '#6b7b94', true);
      drawMountain(1345, 580, 1080, 1620, 980, '#253d63', '#080c14', true);

      ctx.beginPath();
      ctx.moveTo(260, 960);

      ctx.lineTo(1340, 1120);
      ctx.lineTo(260, 1120);
      ctx.closePath();
      ctx.fill();

      if (isHealed) {

        const stones = [

          [1045, 970, 80, 25, -0.08],
          [1145, 955, 70, 22, -0.04]
        ];
        stones.forEach(([sx, sy, sw, sh, rot]) => {
          ctx.save();
          ctx.translate(sx, sy);
          ctx.rotate(rot);
          ctx.fillRect(-sw / 2, -sh / 2, sw, sh);
          ctx.restore();
        });
      }

      // 左侧深蓝前景山坡
      ctx.fillStyle = '#192d4d';
      ctx.strokeStyle = 'rgba(220, 232, 245, 0.65)';
      ctx.lineWidth = 2.2;
      ctx.beginPath();
      ctx.moveTo(0, 935);
      ctx.bezierCurveTo(260, 925, 480, 1060, 760, 1180);
      ctx.lineTo(0, 1180);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();

      // 右侧深蓝前景山坡
      ctx.beginPath();
      ctx.moveTo(1600, 945);
      ctx.bezierCurveTo(1320, 910, 960, 1070, 560, 1180);
      ctx.lineTo(1600, 1180);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
      ctx.restore();

      // 6. 中央白色石质喷泉底座（图1与图2共用）
      ctx.save();
      const fCx = 800;
      const bowlTopY = 805;

      // 喷泉立柱
      const stemGrad = ctx.createLinearGradient(fCx - 75, 0, fCx + 75, 0);
      stemGrad.addColorStop(0, '#cbd5e1');
      stemGrad.addColorStop(0.5, '#f8fafc');
      stemGrad.addColorStop(1, '#94a3b8');
      ctx.fillStyle = stemGrad;
      ctx.beginPath();
      ctx.moveTo(fCx - 52, bowlTopY + 135);
      ctx.quadraticCurveTo(fCx - 45, bowlTopY + 250, fCx - 82, 1120);
      ctx.lineTo(fCx + 82, 1120);
      ctx.quadraticCurveTo(fCx + 45, bowlTopY + 250, fCx + 52, bowlTopY + 135);
      ctx.closePath();
      ctx.fill();

      // 喷泉碗身
      const bowlGrad = ctx.createLinearGradient(fCx - 290, bowlTopY, fCx + 290, bowlTopY + 145);
      bowlGrad.addColorStop(0, '#e2e8f0');
      bowlGrad.addColorStop(0.45, '#f8fafc');
      bowlGrad.addColorStop(1, '#94a3b8');
      ctx.fillStyle = bowlGrad;
      ctx.beginPath();
      ctx.moveTo(fCx - 285, bowlTopY + 28);
      ctx.quadraticCurveTo(fCx - 260, bowlTopY + 145, fCx, bowlTopY + 145);
      ctx.quadraticCurveTo(fCx + 260, bowlTopY + 145, fCx + 285, bowlTopY + 28);
      ctx.closePath();
      ctx.fill();

      ctx.lineWidth = 2;
      ctx.fillRect(fCx - 310, bowlTopY, 620, 16);
      ctx.strokeRect(fCx - 310, bowlTopY, 620, 16);

    /* ==================== 3. 鸟笼整笼连贯左右弯曲扭动引擎 ==================== */
    const cageCanvas = document.getElementById('cage-canvas');
    const cageCtx = cageCanvas.getContext('2d');

    let bendX = 0;          // 当前左右弯曲位移 (px)
    let bendVel = 0;        // 左右晃动角速度
    let hitGlow = 0;        // 受击瞬间发光强度
    let isBroken = false;
    let endingTimer = null;

    // 镜像破碎碎片数组（划破第5次时，将整座鸟笼切割为多边形镜面玻璃碎片飞散）
    let mirrorShards = [];

    // 上下锚定、仅中段连贯弯曲曲线：
    // 顶部穹顶皇冠处 (t <= 0.34) 与底部喷泉碗沿接触处 (t >= 0.67) 偏移严格为 0；
    // 仅笼身中段竖栏 (0.34 < t < 0.67) 随划动产生连续正弦波拱形左右弯曲，绝不上下断开！
    const CAGE_TOP_T = 0.34;
    const CAGE_BOT_T = 0.67;
    function getCageOffsetX(t) {
      if (t <= CAGE_TOP_T || t >= CAGE_BOT_T) return 0;
      const u = (t - CAGE_TOP_T) / (CAGE_BOT_T - CAGE_TOP_T); // 0 (笼肩) -> 1 (笼底)
      return bendX * Math.sin(u * Math.PI);
    }

    // 计算与 CSS background-size: cover 完全一致的绘制矩形，保证鸟笼和背景 100% 对齐
    function getCoverRect(srcW, srcH) {
      const scale = Math.max(W / srcW, H / srcH);
      const drawW = srcW * scale;
      const drawH = srcH * scale;
      return {
        x: (W - drawW) / 2,
        y: (H - drawH) / 2,
        w: drawW,
        h: drawH
      };
    }

    // 生成镜面破碎的多边形碎片（从鸟笼中心呈放射状蜘蛛网裂纹切割整张笼图）
    function createMirrorShatterShards() {
      mirrorShards = [];
      const srcW = cageSource.naturalWidth || cageSource.width;
      const srcH = cageSource.naturalHeight || cageSource.height;
      if (!srcW || !srcH) return;

      const rect = getCoverRect(srcW, srcH);

      // 鸟笼视觉中心坐标（屏幕坐标）
      const centerX = W * 0.5;
      const centerY = rect.y + rect.h * 0.45;

      // 放射状裂纹角度（18 条不规则射线）与 4 圈同心破裂环
      const numRays = 18;
      const angles = [];
      for (let i = 0; i < numRays; i++) {
        angles.push((i / numRays) * Math.PI * 2 + (Math.random() - 0.5) * 0.16);
      }
      angles.sort((a, b) => a - b);

      const maxR = Math.max(rect.w, rect.h) * 0.48;
      const ringRadii = [0, maxR * 0.16, maxR * 0.34, maxR * 0.58, maxR * 0.95];

      // 预生成所有网格交点（加入随机扰动形成尖锐不规则三角/四边形镜面碎片）
      const grid = [];
      for (let rIdx = 0; rIdx < ringRadii.length; rIdx++) {
        const ringPts = [];
        for (let aIdx = 0; aIdx < numRays; aIdx++) {
          if (rIdx === 0) {
            ringPts.push({ x: centerX, y: centerY });
          } else {
            const rJitter = ringRadii[rIdx] * (0.86 + Math.random() * 0.28);
            const aJitter = angles[aIdx] + (Math.random() - 0.5) * 0.08;
            ringPts.push({
              x: centerX + Math.cos(aJitter) * rJitter,
              y: centerY + Math.sin(aJitter) * rJitter
            });
          }
        }
        grid.push(ringPts);
      }

      // 将相邻环与射线围成的区域拆分为尖锐多边形碎片
      for (let rIdx = 0; rIdx < ringRadii.length - 1; rIdx++) {
        for (let aIdx = 0; aIdx < numRays; aIdx++) {
          const nextA = (aIdx + 1) % numRays;
          const p00 = grid[rIdx][aIdx];
          const p01 = grid[rIdx][nextA];
          const p11 = grid[rIdx + 1][nextA];
          const p10 = grid[rIdx + 1][aIdx];

          const poly = rIdx === 0 ? [p00, p10, p11] : [p00, p10, p11, p01];

          // 计算碎片质心
          let cx = 0, cy = 0;
          poly.forEach(pt => { cx += pt.x; cy += pt.y; });
          cx /= poly.length;
          cy /= poly.length;

          // 过滤掉完全处于鸟笼中心区域外的空白碎片，提高渲染帧率
          if (Math.abs(cx - centerX) > rect.w * 0.32 || Math.abs(cy - centerY) > rect.h * 0.38) {
            continue;
          }

          const outAngle = Math.atan2(cy - centerY, cx - centerX) + (Math.random() - 0.5) * 0.35;
          const distRatio = Math.hypot(cx - centerX, cy - centerY) / maxR;
          const speed = (3.5 + (1 - distRatio) * 7.5) * (0.75 + Math.random() * 0.55);

          mirrorShards.push({
            poly,
            cx,
            cy,
            x: cx,
            y: cy,
            vx: Math.cos(outAngle) * speed,
            vy: Math.sin(outAngle) * speed - (1.8 + Math.random() * 2.5),
            gravity: 0.22 + Math.random() * 0.12,
            rot: 0,
            vRot: (Math.random() - 0.5) * 0.16,
            flipAngle: Math.random() * Math.PI,
            vFlip: (Math.random() - 0.5) * 0.14,
            glintPhase: Math.random() * Math.PI * 2,
            life: 1.0,
            decay: 0.011 + Math.random() * 0.008,
            rect
          });
        }
      }
    }

    // 绘制镜面破碎后的多边形碎片（带玻璃反光高光边缘与 3D 翻转透视）
    function renderMirrorShards() {
      if (mirrorShards.length === 0) return;

      for (let i = mirrorShards.length - 1; i >= 0; i--) {
        const s = mirrorShards[i];
        s.x += s.vx;
        s.y += s.vy;
        s.vy += s.gravity;
        s.vx *= 0.985;
        s.rot += s.vRot;
        s.flipAngle += s.vFlip;
        s.glintPhase += 0.22;
        s.life -= s.decay;

        if (s.life <= 0) {
          mirrorShards.splice(i, 1);
          continue;
        }

        cageCtx.save();
        cageCtx.globalAlpha = Math.max(0, s.life);
        cageCtx.translate(s.x, s.y);
        cageCtx.rotate(s.rot);
        // 模拟镜面碎片在空中翻转的 3D 透视缩放
        const scaleX = 0.45 + 0.55 * Math.abs(Math.cos(s.flipAngle));
        cageCtx.scale(scaleX, 1);

        // 构建多边形裁剪路径
        cageCtx.beginPath();
        s.poly.forEach((pt, idx) => {
          const lx = pt.x - s.cx;
          const ly = pt.y - s.cy;
          if (idx === 0) cageCtx.moveTo(lx, ly);
          else cageCtx.lineTo(lx, ly);
        });
        cageCtx.closePath();

        // 1. 裁剪并绘制该碎片对应的鸟笼局部纹理
        cageCtx.save();
        cageCtx.clip();
        cageCtx.drawImage(
          cageSource,
          s.rect.x - s.cx,
          s.rect.y - s.cy,
          s.rect.w,
          s.rect.h
        );

        // 2. 镜面高光反射叠加层（随翻转角度闪烁银白/淡金冷光）
        const glint = Math.max(0, Math.sin(s.glintPhase));
        cageCtx.fillStyle = `rgba(240, 249, 255, ${0.18 + glint * 0.48})`;
        cageCtx.fillRect(-150, -150, 300, 300);
        cageCtx.restore();

        // 3. 锐利玻璃/镜面裂纹亮边描边
        cageCtx.strokeStyle = `rgba(255, 255, 255, ${0.75 * s.life})`;
        cageCtx.lineWidth = 1.4;
        cageCtx.shadowBlur = 10;
        cageCtx.shadowColor = 'rgba(224, 242, 254, 0.9)';
        cageCtx.stroke();

        cageCtx.restore();
      }
    }

    function renderCage() {
      cageCtx.clearRect(0, 0, W, H);

      // 若已触发破笼，则渲染镜面破碎碎片飞散动画
      if (isBroken) {
        renderMirrorShards();
        return;
      }

      // 弹簧阻尼物理更新（仅驱动中段左右弯曲）
      const stiffness = 0.14;
      const damping = 0.82;
      bendVel = (bendVel - stiffness * bendX) * damping;
      bendX += bendVel;
      hitGlow = Math.max(0, hitGlow - 0.055);

      const srcW = cageSource.naturalWidth || cageSource.width;
      const srcH = cageSource.naturalHeight || cageSource.height;
      if (!srcW || !srcH) return;

      const rect = getCoverRect(srcW, srcH);
      const slices = 120; // 120层超细水平切片，确保上下静止、中段连续无缝弯曲
      const srcSliceH = srcH / slices;
      const dstSliceH = rect.h / slices;

      cageCtx.save();

      if (hitGlow > 0.01) {
        cageCtx.filter = `brightness(${1 + hitGlow * 0.95}) drop-shadow(0 0 ${hitGlow * 22}px rgba(254, 240, 138, 0.95))`;
      }

      for (let i = 0; i < slices; i++) {
        const t = i / slices;
        // 顶部皇冠与底部喷泉底座处偏移严格为 0，仅笼身中段随滑动连贯左右弯曲
        const offsetX = getCageOffsetX(t);

        cageCtx.drawImage(
          cageSource,
          0, i * srcSliceH, srcW, srcSliceH + 0.6,
          rect.x + offsetX, rect.y + i * dstSliceH, rect.w, dstSliceH + 0.6
        );
      }

      cageCtx.restore();
    }

    /* ==================== 4. 滑动刀痕、金色火花与 5 次破笼判定 ==================== */
    const REQUIRED_HITS = 5;
    let hits = 0;
    let isDown = false;
    let lastPos = null;
    let strokeDist = 0;
    let countedThisStroke = false;
    let strokeId = 0;

    let trail = [];
    let sparks = [];

    function spawnSparks(x, y, count) {
      for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const spd = 2 + Math.random() * 6.8;
        sparks.push({
          x, y,
          vx: Math.cos(a) * spd,
          vy: Math.sin(a) * spd - 1.5,
          life: 1,
          decay: 0.022 + Math.random() * 0.025,
          size: 1.6 + Math.random() * 2.8
        });
      }
    }

    function renderFX() {
      fxCtx.clearRect(0, 0, W, H);
      fxCtx.save();
      fxCtx.globalCompositeOperation = 'lighter';

      for (let i = trail.length - 1; i >= 0; i--) {
        const p = trail[i];
        p.life -= 0.055;
        if (p.life <= 0) { trail.splice(i, 1); continue; }
        if (i > 0 && trail[i - 1].id === p.id) {
          fxCtx.beginPath();
          fxCtx.moveTo(trail[i - 1].x, trail[i - 1].y);
          fxCtx.lineTo(p.x, p.y);
          fxCtx.lineCap = 'round';
          fxCtx.lineWidth = p.life * 9;
          fxCtx.strokeStyle = `rgba(254, 240, 138, ${p.life * 0.88})`;
          fxCtx.shadowBlur = 18;
          fxCtx.shadowColor = '#facc15';
          fxCtx.stroke();

          fxCtx.lineWidth = p.life * 3;
          fxCtx.strokeStyle = `rgba(255, 255, 255, ${p.life})`;
          fxCtx.stroke();
        }
      }

      for (let i = sparks.length - 1; i >= 0; i--) {
        const s = sparks[i];
        s.x += s.vx;
        s.y += s.vy;
        s.vy += 0.16;
        s.life -= s.decay;
        if (s.life <= 0) { sparks.splice(i, 1); continue; }

        fxCtx.fillStyle = `rgba(254, 243, 199, ${s.life})`;
        fxCtx.shadowBlur = 10;
        fxCtx.shadowColor = '#f59e0b';
        fxCtx.beginPath();
        fxCtx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
        fxCtx.fill();
      }
      fxCtx.restore();
    }

    function loop() {
      renderCage();
      renderFX();
      requestAnimationFrame(loop);
    }

    function triggerBreak() {
      if (isBroken) return;
      isBroken = true;

      try {
        localStorage.setItem('crow_done', 'true');
      } catch (e) {}

      // 1. 生成镜面破碎碎片并向四周爆裂飞散
      createMirrorShatterShards();

      // 2. 屏幕闪一道强光
      const flash = document.getElementById('screen-flash');
      flash.classList.add('active');
      setTimeout(() => flash.classList.remove('active'), 130);

      // 3. 停雨、图1与前景水柱淡出，显露图2（金色月亮、发光乌鸦与归途石板路）
      rainLayer.classList.add('stopped');
      document.getElementById('bg-initial').classList.add('fade-out');
      document.getElementById('fountain-front').classList.add('fade-out');
      document.getElementById('hud-tip').style.opacity = '0';

      spawnSparks(W / 2, H * 0.45, 75);

      // 4. 恢复默认指针并浮现诗句与返回森林按钮
      document.body.style.cursor = 'default';
      customCursor.style.display = 'none';

      endingTimer = setTimeout(() => {
        document.getElementById('ending-panel').classList.add('visible');
      }, 1050);
    }

    function returnToForest() {
      try {
        localStorage.setItem('crow_done', 'true');
      } catch (e) {}

      // 实际项目中取消下面这行注释即可跳回主界面：
      // window.location.href = 'index.html';
      const toast = document.getElementById('toast-msg');
      toast.style.opacity = '1';
      setTimeout(() => { toast.style.opacity = '0'; }, 3200);
    }

    function resetLevel() {
      if (endingTimer) {
        clearTimeout(endingTimer);
        endingTimer = null;
      }
      isBroken = false;
      mirrorShards = [];
      hits = 0;
      bendX = 0;
      bendVel = 0;
      hitGlow = 0;

      sparks = [];

      document.getElementById('hit-num').textContent = '0';
      document.getElementById('hud-tip').style.opacity = '1';
      rainLayer.classList.remove('stopped');

      document.getElementById('fountain-front').classList.remove('fade-out');

    /* ==================== 5. 鼠标与触摸事件绑定 ==================== */
    const customCursor = document.getElementById('custom-cursor');

        /* ==================== 速度拉伸游标物理引擎 ==================== */
    let targetX = window.innerWidth / 2;
    let targetY = window.innerHeight / 2;
    let currentX = targetX;
    let currentY = targetY;
    let currentSpeed = 0;
    let currentAngle = 0;

    window.addEventListener('mousemove', (e) => {
      if (isBroken) return;
      targetX = e.clientX;
      targetY = e.clientY;
    });

    function lerp(start, end, amt) {
      return (1 - amt) * start + amt * end;
    }

    function updateCursorStretch() {
      const customCursor = document.getElementById('custom-cursor');
      if (isBroken) {
        requestAnimationFrame(updateCursorStretch);
        return;
      }

      const prevX = currentX;
      const prevY = currentY;

      currentX = lerp(currentX, targetX, 0.2);
      currentY = lerp(currentY, targetY, 0.2);

      const dx = currentX - prevX;
      const dy = currentY - prevY;
      const instantSpeed = Math.sqrt(dx * dx + dy * dy);

      currentSpeed = lerp(currentSpeed, instantSpeed, 0.15);

      if (instantSpeed > 0.1) {
        const targetAngle = Math.atan2(dy, dx);
        let deltaAngle = targetAngle - currentAngle;
        while (deltaAngle < -Math.PI) deltaAngle += Math.PI * 2;
        while (deltaAngle > Math.PI) deltaAngle -= Math.PI * 2;
        currentAngle += deltaAngle * 0.2;
      }

      const maxSpeed = 30;
      const speedFactor = Math.min(currentSpeed / maxSpeed, 0.75);
      const scaleX = 1 + speedFactor * 0.8;
      const scaleY = 1 - speedFactor * 0.45;

      customCursor.style.transform = `translate3d(${currentX}px, ${currentY}px, 0) translate(-50%, -50%) rotate(${currentAngle}rad) scale(${scaleX}, ${scaleY})`;

      requestAnimationFrame(updateCursorStretch);
    }

    requestAnimationFrame(updateCursorStretch);

    function getPos(e) {
      return e.touches ? { x: e.touches[0].clientX, y: e.touches[0].clientY } : { x: e.clientX, y: e.clientY };
    }

    function onStart(e) {
      if (isBroken) return;
      isDown = true;
      strokeId++;
      strokeDist = 0;
      countedThisStroke = false;
      lastPos = getPos(e);
    }

    function onMove(e) {
      const pos = getPos(e);

      if (!isDown || isBroken || !lastPos) return;
      if (e.cancelable) e.preventDefault();

      const dx = pos.x - lastPos.x;
      const dy = pos.y - lastPos.y;
      trail.push({ x: pos.x, y: pos.y, life: 1, id: strokeId });

      // 判定是否在中央鸟笼区域内划动
      if (Math.abs(pos.x - W / 2) < W * 0.26 && pos.y > H * 0.14 && pos.y < H * 0.72) {
        // 施加连续左右扭动冲量，整笼连贯摇晃
        bendVel += Math.max(-36, Math.min(36, dx)) * 0.48;
        spawnSparks(pos.x, pos.y, 4);

        strokeDist += Math.hypot(dx, dy);
        if (!countedThisStroke && strokeDist >= 75) {
          countedThisStroke = true;
          hits++;
          hitGlow = 1;
          document.getElementById('hit-num').textContent = Math.min(hits, REQUIRED_HITS);
          spawnSparks(pos.x, pos.y, 24);
          if (hits >= REQUIRED_HITS) triggerBreak();
        }
      }
      lastPos = pos;
    }

    function onEnd() {
      isDown = false;
      lastPos = null;
    }

    fxCanvas.addEventListener('mousedown', onStart);
    window.addEventListener('mousemove', onMove);
    window.addEventListener('mouseup', onEnd);
    fxCanvas.addEventListener('touchstart', onStart, { passive: false });
    window.addEventListener('touchmove', onMove, { passive: false });
    window.addEventListener('touchend', onEnd);

    requestAnimationFrame(loop);
  </script>
