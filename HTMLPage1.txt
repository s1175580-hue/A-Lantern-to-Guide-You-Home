    <script>
        // 阶段一逻辑
        const lantern = document.getElementById('lantern-hotspot');
        const whiteFlash = document.getElementById('white-flash');
        const scene1 = document.getElementById('scene-1');
        const scene2 = document.getElementById('scene-2');
        let holdTimer = null;
        let isTransitioning = false;

        function startHold(e) {
        if (isTransitioning) return;
        if (e.cancelable) e.preventDefault();
        lantern.classList.add('charging');
        holdTimer = setTimeout(triggerSceneTransition, 1500);
        }

        function cancelHold() {
        if (isTransitioning) return;
        clearTimeout(holdTimer);
        lantern.classList.remove('charging');
        }

        function triggerSceneTransition() {
        isTransitioning = true;
        whiteFlash.style.opacity = '1';
        setTimeout(() => {
        scene1.style.display = 'none';
        scene2.style.display = 'block';
        whiteFlash.style.opacity = '0';
        setTimeout(() => { isTransitioning = false; }, 1000);
        }, 1000);
        }

        lantern.addEventListener('mousedown', startHold);
        window.addEventListener('mouseup', cancelHold);
        lantern.addEventListener('touchstart', startHold, { passive: false });
        window.addEventListener('touchend', cancelHold);

        // 阶段二逻辑：悬停显示发光图
        const doorUnits = document.querySelectorAll('.door-unit');
        doorUnits.forEach(unit => {
        const hotspot = unit.querySelector('.door-hotspot');
        const animalKey = unit.getAttribute('data-animal');

        hotspot.addEventListener('mouseenter', () => { unit.classList.add('is-hovered'); });
        hotspot.addEventListener('mouseleave', () => { unit.classList.remove('is-hovered'); });

        hotspot.addEventListener('click', () => {
        alert(`准备进入 ${animalKey}.html`);
        // window.location.href = `${animalKey}.html`; // 如果建好了html文件，把这行注释去掉，上面那行删掉
        });
        });

        function skipToScene2() {
        scene1.style.display = 'none';
        scene2.style.display = 'block';
        }
    </script>
